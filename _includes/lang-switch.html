{%- comment -%} Language switch, driven entirely by `site.data.i18n.locales`. The current locale renders as text; the others as links. The default locale's href carries `?lang=` so clicking it overrides a stored preference rather than being bounced straight back by index.html's redirect script. {%- endcomment -%}
{%- for l in site.data.i18n.locales -%} {%- if l.code == page.lang -%} {{ l.label }} {%- else -%} {%- assign target = l.path -%} {%- if page.alt -%}{%- assign target = page.alt[l.code] | default: l.path -%}{%- endif -%} {%- if l.default -%} {%- assign href = target | append: '?lang=' | append: l.code -%} {%- else -%} {%- assign href = target -%} {%- endif -%} {{ l.label }} {%- endif -%} {%- endfor -%}
